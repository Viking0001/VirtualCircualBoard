def printf(number)
    var index = 100
    var draw = 0
    while (index != 0)
        draw = 0
        while (number >= index)
            draw = draw + 1
            number = number - index
        endwhile
        print(draw + 8)
        if (index == 1)
            index = 0
        endif
        if (index == 10)
            index = 1
        endif
        if (index == 100)
            index = 10
        endif
    endwhile
enddef

def drawPixel(x,y)
    console.X = x
    console.Y = y
    print('O')
enddef

def devide(a,b, result)
    var r = 0    
    var aCopy = a
    while (aCopy > b)
        aCopy = aCopy - b
        r = r + 1
    endwhile
    *result = r
enddef

def multiply(a,b,result)
    var r = 0    
    var bCopy = b
    while (bCopy > 0)
        r = r + a
        bCopy = bCopy - 1
    endwhile
    *result = r
enddef
